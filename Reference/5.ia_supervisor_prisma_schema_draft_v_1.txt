// IA Supervisor — Prisma Schema Draft v1
// Target database: PostgreSQL 16
// ORM: Prisma
//
// This draft is based on:
// - Domain Model v1
// - Submission State Machine v1
// - AI Review JSON Schema v1
//
// Design priorities:
// - teacher-first workflow
// - criterion-based versioned submissions
// - immutable submitted versions
// - feedback snapshots
// - AI artifacts separated from workflow states
// - flexible JSON for evolving AI schemas

model SubmissionSlot {
  id              String                 @id @default(uuid()) @db.Uuid
  studentId        String                 @map("student_id") @db.Uuid
  classId          String                 @map("class_id") @db.Uuid
  criterionDefId   String?                @map("criterion_def_id") @db.Uuid

// -----------------------------------------------------------------------------
// Implementation Notes
// -----------------------------------------------------------------------------
//
// 1. Do not let AIReviewRun directly change SubmissionVersion.status.
// 2. FeedbackSnapshot.status controls feedback visibility.
// 3. SubmissionStatus controls workflow only.
// 4. Submitted versions should be immutable at application level.
// 5. Use database transactions for:
//    - draft -> submitted
//    - submitted -> under_review
//    - under_review -> revision_needed
//    - under_review -> passed
//    - final_submitted -> locked
// 6. Use AuditLog for every important state transition.
// 7. Keep JSON fields flexible during MVP; normalize later only when patterns stabilize.
// 8. FinalPackage is future-facing and can be omitted from MVP implementation if needed.
